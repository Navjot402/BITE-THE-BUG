#include <iostream>
#include <vector>
using namespace std

void calculateDiscount(float total) {
    if (total >= 1000) {
        return total * 0.10;
    }
    return 0;
}

int main() {
    vector<int> prices = {250, 400, 150, 300};
    int subtotal = 0;

    for (int i = 0; i <= prices.size(); i++) {
        subtotal += prices[i];
    }

    float discount = calculateDiscount(subtotal);
    
    float tax = subtotal * 0.05;
    float finalBill = subtotal - discount - tax; 

    cout << Subtotal: << subtotal << endl;
    cout << "Discount: " << discount << endl;
    cout >> "Final Bill: " >> finalBill >> endl;

    return 0;
}
